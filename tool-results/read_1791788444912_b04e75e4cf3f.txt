   148→   148→/** 任意方向直箭头（SVG <g>）：p=生长进度 0→1（自根部长出：杆先到、头随之），端点 (x1,y1) 为尖端 */
   149→   149→export const LineArrow: React.FC<{x0: number; y0: number; x1: number; y1: number; p?: number; rodW?: number; headL?: number; headW?: number; color?: string; opacity?: number; dashed?: boolean}> = ({x0, y0, x1, y1, p = 1, rodW = 3, headL = 22, headW = 24, color = WHITE, opacity = 1, dashed = false}) => {
   150→   150→  if (p <= 0) return null;
   151→   151→  const dx = x1 - x0, dy = y1 - y0;
   152→   152→  const L = Math.hypot(dx, dy) || 1;
   153→   153→  const ux = dx / L, uy = dy / L;
   154→   154→  const len = L * clamp01(p);
   155→   155→  const hl = Math.min(headL, len);
   156→   156→  const tx = x0 + ux * len, ty = y0 + uy * len; // 当前尖端
   157→   157→  const bx = tx - ux * hl, by = ty - uy * hl; // 头底中心
   158→   158→  const px = -uy, py = ux;
   159→   159→  const hw = (headW / 2) * (hl / headL);
   160→   160→  return (
   161→   161→    <g opacity={opacity}>
   162→   162→      <line x1={x0} y1={y0} x2={bx + ux * 1} y2={by + uy * 1} stroke={color} strokeWidth={rodW} strokeLinecap="butt" strokeDasharray={dashed ? '8 7' : undefined} />
   163→   163→      <polygon points={`${tx},${ty} ${bx + px * hw},${by + py * hw} ${bx - px * hw},${by - py * hw}`} fill={color} />
   164→   164→    </g>
   165→   165→  );
